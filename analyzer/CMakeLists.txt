cmake_minimum_required(VERSION 3.16)
project(platform_analyzer VERSION 0.4.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(PkgConfig REQUIRED)
pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd>=254)

add_library(analyzer_core src/summary/deltas.cpp src/collectors/linux.cpp src/resource/resource_types.cpp)
target_include_directories(analyzer_core PUBLIC src)
target_compile_options(analyzer_core PRIVATE -Wall -Wextra -Wpedantic)

set(ANALYZER_PCM_LIBRARY "" CACHE FILEPATH "PCM static library built from this project's modified source")
if(ANALYZER_PCM_LIBRARY)
	find_package(Threads REQUIRED)
	target_sources(analyzer_core PRIVATE
		src/collectors/pcm.cpp
		src/collectors/resource_pcm.cpp
		third_party/pcm/src/pcm-iio-pmu.cpp
		third_party/pcm/src/pcm-iio-topology.cpp)
	target_include_directories(analyzer_core PRIVATE third_party/pcm/src)
	target_link_libraries(analyzer_core PUBLIC "${ANALYZER_PCM_LIBRARY}" Threads::Threads)
else()
	target_sources(analyzer_core PRIVATE
		src/collectors/pcm_unavailable.cpp
		src/collectors/resource_pcm_unavailable.cpp)
endif()

add_executable(test_resource tests/test_resource.cpp src/comparison/resource_comparison.cpp)
target_link_libraries(test_resource PRIVATE analyzer_core PkgConfig::SYSTEMD)
target_compile_options(test_resource PRIVATE -UNDEBUG)
add_test(NAME resource_milestone COMMAND test_resource)

add_library(analyzer_session src/session/session.cpp src/batch/batch_record.cpp)
target_link_libraries(analyzer_session PUBLIC analyzer_core PkgConfig::SYSTEMD)
target_compile_definitions(analyzer_session PRIVATE ANALYZER_VERSION="${PROJECT_VERSION}")
target_compile_options(analyzer_session PRIVATE -Wall -Wextra -Wpedantic)
add_executable(test_batch tests/test_batch.cpp src/cli/resource_record.cpp)
target_link_libraries(test_batch PRIVATE analyzer_session analyzer_core PkgConfig::SYSTEMD)
target_compile_options(test_batch PRIVATE -UNDEBUG)
add_test(NAME batch COMMAND test_batch)
add_executable(analyzer src/cli/main.cpp src/cli/resource_record.cpp src/comparison/resource_comparison.cpp)
target_link_libraries(analyzer PRIVATE analyzer_session analyzer_core)
target_compile_options(analyzer PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pmu-perf-check src/cli/pmu_perf_check.cpp)
target_link_libraries(pmu-perf-check PRIVATE analyzer_core)
target_compile_options(pmu-perf-check PRIVATE -Wall -Wextra -Wpedantic)

enable_testing()
add_executable(test_deltas tests/test_deltas.cpp)
target_link_libraries(test_deltas PRIVATE analyzer_core)
add_test(NAME deltas COMMAND test_deltas)
add_executable(test_collectors tests/test_collectors.cpp)
target_link_libraries(test_collectors PRIVATE analyzer_core)
add_test(NAME collectors COMMAND test_collectors)
add_executable(test_session tests/test_session.cpp)
target_link_libraries(test_session PRIVATE analyzer_session)
add_test(NAME session COMMAND test_session)
add_executable(test_pmu tests/test_pmu.cpp)
target_link_libraries(test_pmu PRIVATE analyzer_core)
target_link_options(test_pmu PRIVATE -Wl,--wrap=syscall -Wl,--wrap=read -Wl,--wrap=nanosleep)
add_test(NAME pmu_mock COMMAND test_pmu)
add_executable(test_root_scenario tests/test_root_scenario.cpp)
target_compile_definitions(test_root_scenario PRIVATE ROOT_SCENARIO_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/root-test.sh")
add_test(NAME root_scenario_mock COMMAND test_root_scenario)
add_executable(mock_experiment_analyzer tests/mock_experiment_analyzer.cpp)
add_executable(test_phi4_experiment tests/test_phi4_experiment.cpp)
target_compile_definitions(test_phi4_experiment PRIVATE PHI4_SCENARIO_PATH="${CMAKE_CURRENT_SOURCE_DIR}/experiments/phi4-overhead.sh" MOCK_ANALYZER_PATH="$<TARGET_FILE:mock_experiment_analyzer>")
add_dependencies(test_phi4_experiment mock_experiment_analyzer)
add_test(NAME phi4_experiment_mock COMMAND test_phi4_experiment)
foreach(test_target test_deltas test_collectors test_session test_pmu test_root_scenario test_phi4_experiment mock_experiment_analyzer)
	target_compile_options(${test_target} PRIVATE -UNDEBUG)
endforeach()

if(ANALYZER_PCM_LIBRARY)
	add_executable(pcm-msr-check src/cli/pcm_msr_check.cpp)
	target_include_directories(pcm-msr-check PRIVATE third_party/pcm/src)
	target_link_libraries(pcm-msr-check PRIVATE analyzer_core "${ANALYZER_PCM_LIBRARY}" Threads::Threads)
	target_compile_options(pcm-msr-check PRIVATE -Wall -Wextra -Wpedantic)
	add_executable(pcm_identity tests/pcm_identity.cpp)
	target_include_directories(pcm_identity PRIVATE third_party/pcm/src)
	target_link_libraries(pcm_identity PRIVATE "${ANALYZER_PCM_LIBRARY}" Threads::Threads)
	add_test(NAME pcm_identity COMMAND pcm_identity)
	add_executable(test_pcm_readonly tests/test_pcm_readonly.cpp)
	target_include_directories(test_pcm_readonly PRIVATE third_party/pcm/src)
	target_link_libraries(test_pcm_readonly PRIVATE "${ANALYZER_PCM_LIBRARY}" Threads::Threads)
	target_compile_options(test_pcm_readonly PRIVATE -UNDEBUG)
	target_link_options(test_pcm_readonly PRIVATE -Wl,--wrap=open -Wl,--wrap=__open_2)
	add_test(NAME pcm_readonly COMMAND test_pcm_readonly)
	add_executable(test_pcm_coreonly tests/test_pcm_coreonly.cpp)
	target_include_directories(test_pcm_coreonly PRIVATE third_party/pcm/src)
	target_link_libraries(test_pcm_coreonly PRIVATE "${ANALYZER_PCM_LIBRARY}" Threads::Threads)
	target_compile_options(test_pcm_coreonly PRIVATE -UNDEBUG)
	add_test(NAME pcm_coreonly COMMAND test_pcm_coreonly)
	add_executable(test_pcm_backend tests/test_pcm_backend.cpp)
	target_link_libraries(test_pcm_backend PRIVATE analyzer_core)
	target_compile_options(test_pcm_backend PRIVATE -UNDEBUG)
	target_link_options(test_pcm_backend PRIVATE -Wl,--wrap=access -Wl,--wrap=open -Wl,--wrap=__open_2 -Wl,--wrap=pread -Wl,--wrap=pwrite -Wl,--wrap=syscall -Wl,--wrap=read)
	add_test(NAME pcm_backend_mock COMMAND test_pcm_backend)
endif()
